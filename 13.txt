
# Predict the Package of a Student using Linear Regression with SGD.
import matplotlib.pyplot as plt
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import SGDRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
df = pd.read_csv('/content/placement_cgpa.csv')
X = df.iloc[:, 0:1]
y = df.iloc[:, -1]
plt.scatter(X, y)
plt.xlabel('CGPA')
plt.ylabel('Package (in LPA)')
plt.show()
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=2)
lr = SGDRegressor()
lr.fit(X_train, y_train)
y_pred = lr.predict(X_test)
print("Training R2:", r2_score(y_train, lr.predict(X_train)))
print("Testing R2:", r2_score(y_test, y_pred))
print("MAE:", mean_absolute_error(y_test, y_pred))
print("MSE:", mean_squared_error(y_test, y_pred))
print("RMSE:", np.sqrt(mean_squared_error(y_test, y_pred)))
plt.scatter(X, y)
plt.plot(X_train, lr.predict(X_train), color='red')
plt.xlabel('CGPA')
plt.ylabel('Package (in LPA)')
plt.title("SGD Linear Regression")
plt.show()
m = lr.coef_
b = lr.intercept_
print("Slope (m):", m)
print("Intercept (b):", b)
print("Package for CGPA 8.58:", lr.predict([[8.58]])[0])
print("Package for CGPA 9.5:", lr.predict([[9.5]])[0])
