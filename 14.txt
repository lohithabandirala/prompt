
# Predict House Price using Multiple Linear Regression with Least Squares.
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
X = np.array([[1500, 3, 10], [1800, 4, 15], [2400, 3, 20], [3000, 5, 8], [3500, 4, 12], [4000, 5, 5]])
Y = np.array([300, 400, 350, 500, 450, 550])
model = LinearRegression()
model.fit(X, Y)
Y_pred = model.predict(X)
print("Original Prices:", Y)
print("Predicted Prices:", Y_pred)
print("Intercept:", model.intercept_)
print("Coefficients:", model.coef_)
print("MAE:", mean_absolute_error(Y, Y_pred))
print("MSE:", mean_squared_error(Y, Y_pred))
print("RMSE:", np.sqrt(mean_squared_error(Y, Y_pred)))
print("R2 Score:", r2_score(Y, Y_pred))
print("\nActual vs Predicted Prices:")
for i in range(len(Y)):
    print("Actual:", Y[i] * 1000, "Predicted:", Y_pred[i] * 1000)
plt.scatter(Y, Y_pred, color='blue')
plt.plot([min(Y), max(Y)], [min(Y_pred), max(Y_pred)], color='red')
plt.xlabel("Actual House Prices ($1000s)")
plt.ylabel("Predicted House Prices ($1000s)")
plt.title("Actual vs Predicted House Prices")
plt.show()
new_data = np.array([[2000, 3, 10]])
new_prediction = model.predict(new_data)
print("\nPredicted Price for New House:", new_prediction[0] * 1000)
